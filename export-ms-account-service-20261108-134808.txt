package com.banking.account.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.reactive.EnableWebFluxSecurity;
import org.springframework.security.config.web.server.ServerHttpSecurity;
import org.springframework.security.web.server.SecurityWebFilterChain;

/**
 * Parte III: OAuth2 Resource Server con JWT.
 * Mientras no exista proveedor de identidad, el perfil dev usa permitAll()
 * (regla de docs/common-rules.md: nunca en producción).
 */
@Configuration
@EnableWebFluxSecurity
public class SecurityConfig {

    @Value("${spring.security.dev-mode:false}")
    private boolean devMode;

    @Bean
    public SecurityWebFilterChain securityWebFilterChain(ServerHttpSecurity http) {
        if (devMode) {
            return http
                    .csrf(ServerHttpSecurity.CsrfSpec::disable)
                    .authorizeExchange(exchanges -> exchanges.anyExchange().permitAll())
                    .build();
        }
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2.jwt(jwt -> { }))
                .build();
    }
}

package com.banking.account.controller;

import com.banking.account.api.AccountApi;
import com.banking.account.dto.AccountCreateRequest;
import com.banking.account.dto.AccountPage;
import com.banking.account.dto.AccountResponse;
import com.banking.account.dto.AccountUpdateRequest;
import com.banking.account.dto.BalanceResponse;
import com.banking.account.dto.MovementPage;
import com.banking.account.dto.OperationAcceptedResponse;
import com.banking.account.dto.TransferRequest;
import com.banking.account.service.AccountService;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.PageRequest;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Flux;
import reactor.core.publisher.Mono;

import java.util.UUID;

@RestController
@RequiredArgsConstructor
public class AccountApiImpl implements AccountApi {

    private final AccountService accountService;

    @Override
    public Mono<ResponseEntity<AccountPage>> findAllAccounts(
            String xCorrelationID, String xChannelID, Integer page, Integer size,
            ServerWebExchange exchange) {
        // Listado paginado simple del servicio (contrato: AccountPage).
        return accountService.listAccounts(page, size)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<AccountResponse>> createAccount(
            String xCorrelationID, String xChannelID, UUID xTransactionID,
            Mono<AccountCreateRequest> accountCreateRequest, ServerWebExchange exchange) {
        return accountCreateRequest
                .flatMap(req -> accountService.create(req, xTransactionID.toString()))
                .map(response -> ResponseEntity.status(HttpStatus.CREATED).body(response));
    }

    @Override
    public Mono<ResponseEntity<AccountResponse>> findAccountById(
            UUID accountId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return accountService.findById(accountId).map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<AccountResponse>> updateAccount(
            UUID accountId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, Mono<AccountUpdateRequest> accountUpdateRequest,
            ServerWebExchange exchange) {
        return accountUpdateRequest
                .flatMap(req -> accountService.update(accountId, req))
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<Void>> deactivateAccount(
            UUID accountId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, ServerWebExchange exchange) {
        return accountService.deactivate(accountId)
                .thenReturn(ResponseEntity.noContent().<Void>build());
    }

    @Override
    public Mono<ResponseEntity<MovementPage>> findAccountMovements(
            UUID accountId, String xCorrelationID, String xChannelID,
            Integer page, Integer size, ServerWebExchange exchange) {
        return accountService.findMovements(accountId, page, size).map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<BalanceResponse>> findAccountBalance(
            UUID accountId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return accountService.findBalance(accountId).map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<OperationAcceptedResponse>> createTransfer(
            String xCorrelationID, String xChannelID, UUID xTransactionID,
            Mono<TransferRequest> transferRequest, ServerWebExchange exchange) {
        return transferRequest
                .flatMap(req -> accountService.transfer(req, xTransactionID.toString()))
                .map(response -> ResponseEntity.status(HttpStatus.ACCEPTED).body(response));
    }
}

package com.banking.account.exception;

import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.web.reactive.error.ErrorWebExceptionHandler;
import org.springframework.context.annotation.Configuration;
import org.springframework.core.Ordered;
import org.springframework.core.io.buffer.DataBuffer;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.List;

@Slf4j
@Configuration
@RequiredArgsConstructor
public class GlobalExceptionHandler implements ErrorWebExceptionHandler, Ordered {

    private final ObjectMapper objectMapper;

    @Override
    public int getOrder() {
        return -2;
    }

    @Override
    public Mono<Void> handle(ServerWebExchange exchange, Throwable ex) {
        HttpStatus status;
        String code;
        String message;

        if (ex instanceof DomainException domainException) {
            status = domainException.getStatus();
            code = domainException.getCode();
            message = domainException.getMessage();

            if (status.is5xxServerError()) {
                log.error("Dependencia o error de servidor en {}: {} - {}",
                        exchange.getRequest().getPath(), code, message);
            } else {
                log.warn("Regla de negocio rechazada en {}: {} - {}",
                        exchange.getRequest().getPath(), code, message);
            }
        } else {
            status = HttpStatus.INTERNAL_SERVER_ERROR;
            code = "INTERNAL_ERROR";
            // No exponer ex.getMessage() al cliente en errores inesperados.
            message = "Unexpected error";
            log.error("Error inesperado procesando {}", exchange.getRequest().getPath(), ex);
        }

        String correlationId = exchange.getRequest().getHeaders()
                .getFirst("X-Correlation-ID");

        ErrorResponse error = ErrorResponse.builder()
                .code(code)
                .message(message)
                .timestamp(Instant.now())
                .correlationId(correlationId)
                .details(List.of())
                .build();

        byte[] bytes;
        try {
            bytes = objectMapper.writeValueAsBytes(error);
        } catch (Exception serializationError) {
            log.error("No se pudo serializar ErrorResponse", serializationError);
            bytes = ("{\"code\":\"" + code + "\",\"message\":\""
                    + escape(message) + "\"}")
                    .getBytes(StandardCharsets.UTF_8);
        }

        exchange.getResponse().setStatusCode(status);
        exchange.getResponse().getHeaders().setContentType(MediaType.APPLICATION_JSON);
        DataBuffer buffer = exchange.getResponse().bufferFactory().wrap(bytes);
        return exchange.getResponse().writeWith(Mono.just(buffer));
    }

    private String escape(String s) {
        return s == null ? "" : s.replace("\"", "\\\"");
    }
}

@Slf4j
@Service
@RequiredArgsConstructor
public class AccountServiceImpl implements AccountService {

    private final AccountRepository accountRepository;
    private final MovementRepository movementRepository;
    private final AccountMapper mapper;
    private final AccountProperties properties;
    private final CardVerificationClient cardVerificationClient;
    private final AccountEventPublisher eventPublisher;
    private final TransferFacade transferFacade;
    private final ObjectMapper objectMapper;

    @Override
    public Mono<AccountResponse> create(AccountCreateRequest request, String transactionId) {
        return validateCreationRules(request)
                .then(generateAccountNumber(request.getCurrency()))
                .map(accountNumber -> buildDocument(request, accountNumber))
                .flatMap(accountRepository::save)
                .flatMap(saved -> registerOpeningMovementIfAny(saved, request)
                        .then(publishEvent("AccountCreated", saved))
                        .thenReturn(mapper.toResponse(saved)));
    }

    private Mono<Void> validateCreationRules(AccountCreateRequest request) {
        AccountType type = request.getAccountType();
        com.banking.account.dto.AccountProfile profile = request.getProfile();

        Mono<Void> businessRules = Mono.defer(() -> {
            if (type == AccountType.SAVINGS || type == AccountType.FIXED_TERM) {
                if (profile == com.banking.account.dto.AccountProfile.PYME) {
                    return Mono.error(new UnprocessableEntityException(
                            "PYME profile only allows CHECKING accounts"));
                }
            }
            if (request.getInitialAmount() != null) {
                BigDecimal min = minimumOpening(profile);
                if (new BigDecimal(request.getInitialAmount()).compareTo(min) < 0) {
                    return Mono.error(new UnprocessableEntityException(
                            "Initial amount below configured minimum opening amount " + min));
                }
            }
            if (profile == com.banking.account.dto.AccountProfile.VIP
                    || profile == com.banking.account.dto.AccountProfile.PYME) {
                return cardVerificationClient.hasCreditCard(request.getCustomerId())
                        .flatMap(has -> Boolean.TRUE.equals(has) ? Mono.empty()
                                : Mono.error(new UnprocessableEntityException(
                                "Profile " + profile + " requires a bank credit card")));
            }
            return Mono.empty();
        });

        Mono<Void> typeRules = Mono.defer(() -> {
            if (type == AccountType.SAVINGS) {
                return accountRepository.findByCustomerIdAndAccountType(
                                request.getCustomerId(), AccountType.SAVINGS.name())
                        .hasElements()
                        .flatMap(exists -> exists
                                ? Mono.error(new ConflictException(
                                "PERSONAL customer already has a SAVINGS account"))
                                : Mono.empty());
            }
            return Mono.empty();
        });

        return businessRules.then(typeRules);
    }

    private BigDecimal minimumOpening(com.banking.account.dto.AccountProfile profile) {
        AccountProperties.MinimumOpening min = properties.getMinimumOpeningAmount();
        return switch (profile) {
            case VIP -> min.getVip();
            case PYME -> min.getPyme();
            default -> min.getStandard();
        };
    }

    private AccountDocument buildDocument(AccountCreateRequest request, String accountNumber) {
        return AccountDocument.builder()
                .id(UUID.randomUUID())
                .accountNumber(accountNumber)
                .customerId(request.getCustomerId())
                .accountType(request.getAccountType().name())
                .profile(request.getProfile().name())
                .status(AccountStatus.ACTIVE.name())
                .balance(request.getInitialAmount() == null
                        ? BigDecimal.ZERO : new BigDecimal(request.getInitialAmount()))
                .currency(request.getCurrency())
                .holders(request.getHolders() == null ? List.of() : request.getHolders())
                .signers(request.getSigners() == null ? List.of() : request.getSigners())
                .monthlyMovements(0)
                .monthlyMovementsMonth(AccountDocument.currentMonth())
                .createdAt(Instant.now())
                .build();
    }

    private Mono<String> generateAccountNumber(String currency) {
        return Mono.fromSupplier(() -> {
            String number = String.format("%s-%013d",
                    currency.equals("USD") ? "002" : "001",
                    Math.abs(UUID.randomUUID().getLeastSignificantBits()) % 10_000_000_000_000L);
            return number;
        });
    }

    private Mono<Void> registerOpeningMovementIfAny(AccountDocument account,
                                                    AccountCreateRequest request) {
        if (request.getInitialAmount() == null
                || new BigDecimal(request.getInitialAmount()).compareTo(BigDecimal.ZERO) == 0) {
            return Mono.empty();
        }
        MovementDocument movement = MovementDocument.builder()
                .id(UUID.randomUUID())
                .accountId(account.getId())
                .type("DEPOSIT")
                .amount(new BigDecimal(request.getInitialAmount()))
                .balanceAfter(account.getBalance())
                .timestamp(Instant.now())
                .build();
        return movementRepository.save(movement).then();
    }

    private Mono<Void> publishEvent(String eventType, AccountDocument account) {
        return Mono.fromRunnable(() -> {
            try {
                String payload = objectMapper.writeValueAsString(java.util.Map.of(
                        "eventType", eventType,
                        "accountId", String.valueOf(account.getId()),
                        "customerId", String.valueOf(account.getCustomerId()),
                        "accountType", account.getAccountType()));
                eventPublisher.publish(eventType, payload);
            } catch (Exception ex) {
                log.warn("No se pudo serializar evento {}: {}", eventType, ex.getMessage());
            }
        });
    }

    @Override
    public Mono<AccountResponse> findById(UUID accountId) {
        return accountRepository.findById(accountId)
                .switchIfEmpty(Mono.error(new NotFoundException("Account not found: " + accountId)))
                .map(mapper::toResponse);
    }

    @Override
    public Mono<AccountResponse> update(UUID accountId, AccountUpdateRequest request) {
        return accountRepository.findById(accountId)
                .switchIfEmpty(Mono.error(new NotFoundException("Account not found: " + accountId)))
                .map(account -> {
                    if (request.getProfile() != null) {
                        account.setProfile(request.getProfile().name());
                    }
                    if (request.getStatus() != null) {
                        account.setStatus(request.getStatus().name());
                    }
                    return account;
                })
                .flatMap(accountRepository::save)
                .flatMap(saved -> publishEvent("AccountUpdated", saved)
                        .thenReturn(mapper.toResponse(saved)));
    }

    @Override
    public Mono<Void> deactivate(UUID accountId) {
        return accountRepository.findById(accountId)
                .switchIfEmpty(Mono.error(new NotFoundException("Account not found: " + accountId)))
                .flatMap(account -> {
                    if (account.getBalance() != null
                            && account.getBalance().compareTo(BigDecimal.ZERO) != 0) {
                        return Mono.error(new ConflictException(
                                "Cannot deactivate an account with a non-zero balance"));
                    }
                    account.setStatus(AccountStatus.INACTIVE.name());
                    return accountRepository.save(account)
                            .flatMap(saved -> publishEvent("AccountDeactivated", saved))
                            .then();
                });
    }

    @Override
    public Mono<MovementPage> findMovements(UUID accountId, int page, int size) {
        return accountRepository.existsById(accountId)
                .flatMap(exists -> {
                    if (!exists) {
                        return Mono.error(new NotFoundException("Account not found: " + accountId));
                    }
                    Pageable pageable = PageRequest.of(page, size);
                    Mono<List<MovementResponse>> content =
                            movementRepository.findByAccountIdOrderByTimestampDesc(accountId, pageable)
                                    .map(mapper::toResponse)
                                    .collectList();
                    Mono<Long> total = movementRepository.countByAccountId(accountId);
                    return Mono.zip(content, total)
                            .map(tuple -> {
                                MovementPage movementPage = new MovementPage();
                                movementPage.setContent(tuple.getT1());
                                movementPage.setPage(page);
                                movementPage.setSize(size);
                                movementPage.setTotalElements(tuple.getT2().intValue());
                                return movementPage;
                            });
                });
    }

    @Override
    public Mono<BalanceResponse> findBalance(UUID accountId) {
        return accountRepository.findById(accountId)
                .switchIfEmpty(Mono.error(new NotFoundException("Account not found: " + accountId)))
                .map(account -> {
                    BalanceResponse response = new BalanceResponse();
                    response.setAccountId(account.getId());
                    response.setAvailableBalance(account.getBalance() == null
                            ? "0.00" : account.getBalance().toPlainString());
                    response.setCurrency(account.getCurrency());
                    response.setLastUpdated(account.getCreatedAt() == null
                            ? OffsetDateTimeMin()
                            : java.time.OffsetDateTime.ofInstant(account.getCreatedAt(), ZoneOffset.UTC));
                    return response;
                });
    }

    private java.time.OffsetDateTime OffsetDateTimeMin() {
        return java.time.OffsetDateTime.ofInstant(Instant.EPOCH, ZoneOffset.UTC);
    }

    public Mono<OperationAcceptedResponse> transfer(TransferRequest request, String transactionId) {
        return transferFacade.transfer(request, transactionId);
    }

    @Override

        Mono<List<AccountResponse>> content = accountRepository.findAll()
                .skip((long) page * size)
                .take(size)
                .map(mapper::toResponse)
                .collectList();
        return accountRepository.count().zipWith(content)
                .map(tuple -> {
                    AccountPage accountPage = new AccountPage();
                    accountPage.setContent(tuple.getT2());
                    accountPage.setPage(page);
                    accountPage.setSize(size);
                    accountPage.setTotalElements(tuple.getT1().intValue());
                    return accountPage;
                });
    }

import com.banking.account.config.AccountProperties;

import java.time.ZoneOffset;

/**
 * Cálculo de comisiones por movimiento y reglas de cuenta FIXED_TERM.
 */
@Component
public class MovementFeeCalculator {

    private final AccountProperties properties;

    public MovementFeeCalculator(AccountProperties properties) {
        this.properties = properties;
    }

    /**
     * Regla FIXED_TERM: solo un movimiento (depósito o retiro) en el día del mes.
     * Segundo movimiento el mismo día → 422.
     */
    public void validateFixedTerm(AccountDocument account) {
        if (!"FIXED_TERM".equals(account.getAccountType())) {
            return;
        }
        int today = LocalDate.now(ZoneOffset.UTC).getDayOfMonth();
        if (account.getLastMovementDayOfMonth() != null
                && account.getLastMovementDayOfMonth() == today) {
            throw new UnprocessableEntityException(
                    "FIXED_TERM account allows only one movement per month on its day");
        }
    }

    /**
     * Comisión por movimiento excedente:
     * - Si la cuenta supera el cupo mensual de movimientos de su tipo → comisión extra.
     * - Si supera el número de depósitos/retiros gratuitos → comisión por transacción.
     */
    public BigDecimal feeFor(AccountDocument account, boolean isWithdrawal) {
        BigDecimal fee = BigDecimal.ZERO;

        AccountProperties.Savings savings = properties.getSavings();
        AccountProperties.Checking checking = properties.getChecking();

        if ("SAVINGS".equals(account.getAccountType())
                && savings.getMaxMonthlyMovements() != null
                && account.getMonthlyMovements() >= savings.getMaxMonthlyMovements()) {
            fee = fee.add(savings.getExtraMovementFee());
        } else if ("CHECKING".equals(account.getAccountType())
                && checking.getMaxMonthlyMovements() != null
                && account.getMonthlyMovements() >= checking.getMaxMonthlyMovements()) {
            fee = fee.add(checking.getExtraMovementFee());
        }

        AccountProperties.Transactions tx = properties.getTransactions();
        int freeLimit = isWithdrawal
                ? tx.getMaxFreeWithdrawals() : tx.getMaxFreeDeposits();
        if (account.getMonthlyMovements() >= freeLimit) {
            fee = fee.add(tx.getPerTransactionFee());
        }
        return fee;
    }
}

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.core.publisher.Mono;

import java.math.BigDecimal;
import java.time.Duration;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

/**
 * Fachada de transferencias: delega la orquestación a ms-transaction-service.
 * No mueve saldo localmente.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class TransferFacade {

    private final AccountRepository accountRepository;
    private final WebClient.Builder webClientBuilder;
    private final AccountEventPublisher eventPublisher;
    private final ObjectMapper objectMapper;
    private final AccountProperties properties;

    @Value("${banking.transaction-service.base-url:http://ms-transaction-service}")
    private String transactionServiceBaseUrl;

    @TimeLimiter(name = "transactionService")
    public Mono<com.banking.account.dto.OperationAcceptedResponse> transfer(
            TransferRequest request, String transactionId) {

        if (request.getSourceAccountId().equals(request.getTargetAccountId())) {
            return Mono.error(new com.banking.account.exception.UnprocessableEntityException(
                    "sourceAccountId and targetAccountId must be different"));
        }

        Mono<AccountDocument> source = accountRepository.findById(request.getSourceAccountId())
                .switchIfEmpty(Mono.error(new NotFoundException(
                        "Source account not found: " + request.getSourceAccountId())));

        Mono<AccountDocument> target = accountRepository.findById(request.getTargetAccountId())
                .switchIfEmpty(Mono.error(new NotFoundException(
                        "Target account not found: " + request.getTargetAccountId())));

        return Mono.zip(source, target)
                .flatMap(tuple -> validate(tuple.getT1(), tuple.getT2(), request))
                .then(callTransactionService(request, transactionId));
    }

    private Mono<Void> validate(AccountDocument source, AccountDocument target,
                                TransferRequest request) {
        return Mono.defer(() -> {
            if (!"ACTIVE".equals(source.getStatus())) {
                return Mono.error(new ConflictException(
                        "Source account is " + source.getStatus()));
            }
            if (!"ACTIVE".equals(target.getStatus())) {
                return Mono.error(new ConflictException(
                        "Target account is not ACTIVE"));
            }
            BigDecimal amount = new BigDecimal(request.getAmount());
            if (source.getBalance() == null || source.getBalance().compareTo(amount) < 0) {
                return Mono.error(new ConflictException("Insufficient funds"));
            }
            return Mono.empty();
        });
    }

    private Mono<com.banking.account.dto.OperationAcceptedResponse> callTransactionService(
            TransferRequest request, String transactionId) {
        return webClientBuilder.baseUrl(transactionServiceBaseUrl).build()
                .post()
                .uri("/api/v1/transactions")
                .header("X-Transaction-ID", transactionId)
                .header("X-Channel-ID", "ACCOUNT_FACADE")
                .bodyValue(toTransactionPayload(request, transactionId))
                .retrieve()
                .bodyToMono(Map.class)
                .timeout(Duration.ofSeconds(30))
                .map(response -> {
                    com.banking.account.dto.OperationAcceptedResponse accepted =
                            new com.banking.account.dto.OperationAcceptedResponse();
                    Object operationId = response.get("transactionId");
                    accepted.setOperationId(operationId != null
                            ? UUID.fromString(String.valueOf(operationId)) : UUID.randomUUID());
                    accepted.setStatus(com.banking.account.dto.OperationAcceptedResponse.StatusEnum.ACCEPTED);
                    accepted.setMessage("Transfer accepted for processing by Transaction Service");
                    return accepted;
                })
                .onErrorResume(ex -> {
                    log.error("Error delegating to transaction-service: {}", ex.getMessage());
                    return Mono.error(new com.banking.account.exception.ConflictException(
                            "Transaction Service is unavailable, transfer not accepted"));
                });
    }

    private Map<String, Object> toTransactionPayload(TransferRequest request, String transactionId) {
        Map<String, Object> payload = new HashMap<>();
        payload.put("transactionId", transactionId);
        payload.put("type", "TRANSFER");
        payload.put("sourceAccountId", String.valueOf(request.getSourceAccountId()));
        payload.put("targetAccountId", String.valueOf(request.getTargetAccountId()));
        payload.put("amount", request.getAmount());
        payload.put("currency", request.getCurrency());
        if (request.getDescription() != null) {
            payload.put("description", request.getDescription());
        }
        return payload;
    }
}

import reactor.test.StepVerifier;

import java.util.UUID;

class AccountServiceImplRulesTest {

    private static final UUID CUSTOMER_ID =
            UUID.fromString("5eb3b5a0-9c11-4dbf-b5a1-4a1e5b1a2a01");

    private AccountRepository accountRepository;
    private MovementRepository movementRepository;
    private CardVerificationClient cardClient;
    private AccountServiceImpl service;

    @BeforeEach
    void setUp() {
        accountRepository = Mockito.mock(AccountRepository.class);
        movementRepository = Mockito.mock(MovementRepository.class);
        cardClient = Mockito.mock(CardVerificationClient.class);
        AccountMapper mapper = new AccountMapper();
        AccountProperties properties = new AccountProperties();
        AccountEventPublisher publisher = (type, payload) -> { };
        TransferFacade transferFacade = Mockito.mock(TransferFacade.class);

        service = new AccountServiceImpl(accountRepository, movementRepository, mapper,
                properties, cardClient, publisher, transferFacade, new ObjectMapper());
    }

    private AccountCreateRequest request(AccountType type, AccountProfile profile) {
        AccountCreateRequest request = new AccountCreateRequest();
        request.setCustomerId(CUSTOMER_ID);
        request.setAccountType(type);
        request.setProfile(profile);
        request.setCurrency("PEN");
        return request;
    }

    @Test
    void personalSecondSavingsConflicts() {
        Mockito.when(accountRepository.findByCustomerIdAndAccountType(
                        CUSTOMER_ID, AccountType.SAVINGS.name()))
                .thenReturn(Flux.just(new AccountDocument()));

        StepVerifier.create(service.create(request(AccountType.SAVINGS,
                        AccountProfile.STANDARD), "tx-1"))
                .expectError(ConflictException.class)
                .verify();
    }

    @Test
    void vipWithoutCreditCardFails() {
        Mockito.when(accountRepository.findByCustomerIdAndAccountType(
                        Mockito.any(), Mockito.any()))
                .thenReturn(Flux.empty());
        Mockito.when(cardClient.hasCreditCard(CUSTOMER_ID)).thenReturn(Mono.just(false));

        StepVerifier.create(service.create(request(AccountType.SAVINGS,
                        AccountProfile.VIP), "tx-2"))
                .expectError(UnprocessableEntityException.class)
                .verify();
    }

    @Test
    void pymeWithoutCreditCardFails() {
        Mockito.when(cardClient.hasCreditCard(CUSTOMER_ID)).thenReturn(Mono.just(false));

        StepVerifier.create(service.create(request(AccountType.CHECKING,
                        AccountProfile.PYME), "tx-3"))
                .expectError(UnprocessableEntityException.class)
                .verify();
    }

    @Test
    void pymeWithSavingsTypeFails() {
        StepVerifier.create(service.create(request(AccountType.SAVINGS,
                        AccountProfile.PYME), "tx-4"))
                .expectError(UnprocessableEntityException.class)
                .verify();
    }

    @Test
    void vipWithCreditCardSucceeds() {
        Mockito.when(accountRepository.findByCustomerIdAndAccountType(
                        Mockito.any(), Mockito.any()))
                .thenReturn(Flux.empty());
        Mockito.when(cardClient.hasCreditCard(CUSTOMER_ID)).thenReturn(Mono.just(true));
        Mockito.when(accountRepository.save(Mockito.any(AccountDocument.class)))
                .thenAnswer(invocation -> Mono.just(invocation.getArgument(0)));
        Mockito.when(movementRepository.save(Mockito.any()))
                .thenReturn(Mono.just(new com.banking.account.document.MovementDocument()));

        StepVerifier.create(service.create(request(AccountType.SAVINGS,
                        AccountProfile.VIP), "tx-5"))
                .expectNextCount(1)
                .verifyComplete();
    }
}

package com.banking.account.service.impl;

import com.banking.account.config.AccountProperties;
import com.banking.account.document.AccountDocument;
import com.banking.account.exception.UnprocessableEntityException;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.math.BigDecimal;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertTrue;

class MovementFeeCalculatorTest {

    private MovementFeeCalculator calculator;
    private AccountProperties properties;

    @BeforeEach
    void setUp() {
        properties = new AccountProperties();
        properties.getSavings().setMaxMonthlyMovements(5);
        properties.getSavings().setExtraMovementFee(new BigDecimal("2.00"));
        properties.getTransactions().setMaxFreeDeposits(5);
        properties.getTransactions().setMaxFreeWithdrawals(5);
        properties.getTransactions().setPerTransactionFee(new BigDecimal("1.00"));
        calculator = new MovementFeeCalculator(properties);
    }

    private AccountDocument account(String type, int monthlyMovements) {
        AccountDocument account = AccountDocument.builder().build();
        account.setAccountType(type);
        account.setMonthlyMovements(monthlyMovements);
        return account;
    }

    @Test
    void savingsWithinFreeQuotaNoFee() {
        assertTrue(calculator.feeFor(account("SAVINGS", 0), false)
                .compareTo(BigDecimal.ZERO) == 0);
    }

    @Test
    void savingsExceedingMonthlyQuotaChargesExtraFee() {
        BigDecimal fee = calculator.feeFor(account("SAVINGS", 5), false);
        // 2.00 por superar el límite mensual + 1.00 por superar cupo gratuito
        assertEquals(0, fee.compareTo(new BigDecimal("3.00")));
    }

    @Test
    void checkingHasNoMonthlyLimitByDefault() {
        assertTrue(calculator.feeFor(account("CHECKING", 1000), false)
                .compareTo(new BigDecimal("1.00")) == 0);
    }

    @Test
    void fixedTermSecondMovementSameDayFails() {
        AccountDocument fixedTerm = account("FIXED_TERM", 0);
        fixedTerm.setLastMovementDayOfMonth(java.time.LocalDate.now(
                java.time.ZoneOffset.UTC).getDayOfMonth());
        assertThrows(UnprocessableEntityException.class,
                () -> calculator.validateFixedTerm(fixedTerm));
    }

    @Test
    void fixedTermMovementOnDifferentDayAllowed() {
        AccountDocument fixedTerm = account("FIXED_TERM", 0);
        fixedTerm.setLastMovementDayOfMonth(null);
        calculator.validateFixedTerm(fixedTerm);
        fixedTerm.setLastMovementDayOfMonth(1);
        if (java.time.LocalDate.now(java.time.ZoneOffset.UTC).getDayOfMonth() != 1) {
            calculator.validateFixedTerm(fixedTerm);
        }
    }
}
